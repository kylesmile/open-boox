# Built-in packages for scripts/debloat.sh, checked on a Note Air 4C,
# firmware 4.2-rel_04282.
#
#   keep    leave installed
#   remove  remove for user 0 (the APK stays on /system; revert brings it back)
#   decide  left alone until you pick keep or remove (scripts/debloat.sh pick)
#
# Your picks go in debloat/choices.txt, which is not committed and wins over
# this file. Packages in neither file are never touched. The launcher and the
# keyboard in use are never removed, whatever these files say.
#
# action  package                                  description

# Boox apps the tablet needs
keep      com.onyx                                 Launcher, library and Boox settings
keep      com.onyx.floatingbutton                  Floating button (refresh modes)
keep      com.onyx.tscalibration                   Pen and touch calibration
keep      com.onyx.android.note                    Notes

# Boox extras

remove    com.onyx.clock                           Clock

remove    com.onyx.easytransfer                    Easy transfer
